Install
pip install python-jose
Generate JWT
from jose import jwt
from datetime import datetime
from datetime import timedelta

SECRET_KEY = "mysecret"

ALGORITHM = "HS256"

def create_token(username):

    expire = (
        datetime.utcnow()
        + timedelta(minutes=30)
    )

    payload = {
        "sub": username,
        "exp": expire
    }

    return jwt.encode(
        payload,
        SECRET_KEY,
        algorithm=ALGORITHM
    )
Decode JWT
payload = jwt.decode(

class UserLogin(BaseModel):

    username: str

Login API
from fastapi import FastAPI
from fastapi import HTTPException

app = FastAPI()

@app.post("/login")
def login(user: UserLogin):

    if (
        user.username != "admin"
        or
        user.password != "admin123"
    ):
        raise HTTPException(
            401,
            "Invalid Credentials"
        )

    token = create_token(
        user.username
    )

    return {
        "access_token": token
    }

Current User Dependency
from fastapi import Depends

def get_current_user(
        token: str = Depends(
            oauth2_scheme
        )
):

    payload = jwt.decode(
        token,
        SECRET_KEY,
        algorithms=["HS256"]
    )

    return payload["sub"]
Protected API
@app.get("/profile")
def profile(
    current_user=Depends(
        get_current_user
    )
):

    return {
        "user": current_user
    }
