Session 1 (6 AM – 8 AM)
Create Customer API
Customer Model
from sqlalchemy import Column
from sqlalchemy import Integer
from sqlalchemy import String

from database.database import Base

class Customer(Base):

    __tablename__ = "customers"

    id = Column(
        Integer,
        primary_key=True,
        index=True
    )

    name = Column(
        String,
        nullable=False
    )

    email = Column(
        String,
        unique=True
    )
Request Schema
from pydantic import BaseModel
from pydantic import EmailStr

class CustomerCreate(
    BaseModel
):
    name: str
    email: EmailStr
Response Schema
class CustomerResponse(
    BaseModel
):

    id: int
    name: str
    email: str

    class Config:
        from_attributes = True
Create API
@router.post(
    "/customers",
    response_model=
    CustomerResponse
)
def create_customer(
    request: CustomerCreate,
    db: Session =
    Depends(get_db)
):

    customer = Customer(
        name=request.name,
        email=request.email
    )

    db.add(customer)

    db.commit()

    db.refresh(customer)

    return customer

Get Customer By Id
@router.get(
    "/customers/{id}"
)
def get_customer(
    id: int,
    db: Session =
    Depends(get_db)
):

    customer = (
        db.query(Customer)
        .filter(
            Customer.id == id
        )
        .first()
    )

    return customer

Session 3 (4 PM – 6 PM)
Update & Delete
Update Customer
@router.put(
    "/customers/{id}"
)
def update_customer(
        id: int,
        request:
        CustomerCreate,

        db: Session =
        Depends(get_db)
):

    customer = (
        db.query(Customer)
        .filter(
            Customer.id == id
        )
        .first()
    )

    if not customer:

        raise HTTPException(
            404,
            "Customer Not Found"
        )

    customer.name = (
        request.name
    )

    customer.email = (
        request.email
    )

    db.commit()

    db.refresh(customer)

    return customer
Delete Customer
@router.delete(
    "/customers/{id}"
)
def delete_customer(
        id: int,
        db: Session =
        Depends(get_db)
):

    customer = (
        db.query(Customer)
        .filter(
            Customer.id == id
        )
        .first()
    )

    if not customer:

        raise HTTPException(
            404,
            "Customer Not Found"
        )

    db.delete(customer)

    db.commit()

    return {
        "message":
        "Deleted Successfully"
    }
Service Layer Pattern

Avoid putting business logic in routers.

customer_service.py
class CustomerService:

    @staticmethod
    def create_customer(
        request,
        db
    ):

        customer = Customer(
            name=request.name,
            email=request.email
        )

        db.add(customer)

        db.commit()

        db.refresh(customer)

        return customer
Router
return (
    CustomerService
    .create_customer(
        request,
        db
    )
)

Pagination

Bad:

db.query(Customer).all()

Good:

customers = (
    db.query(Customer)
      .offset(skip)
      .limit(limit)
      .all()
)
