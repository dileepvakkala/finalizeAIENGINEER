Session 3 (Implementation)
PDF Processing Service
utils/pdf_processor.py
from pypdf import PdfReader


def extract_text(
        file_path
):

    reader = PdfReader(
        file_path
    )

    text = ""

    for page in reader.pages:

        text += (
            page.extract_text()
            or ""
        )

    return text
Processing Service

Delete Document
@router.delete("/{id}")
def delete_document(
        id: int,
        db: Session
):

    document = db.query(
        Document
    ).filter(
        Document.id == id
    ).first()

    os.remove(
        document.filepath
    )

    db.delete(document)

    db.commit()

    return {
        "message":
        "Deleted"
    }
